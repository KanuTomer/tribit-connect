cmake_minimum_required(VERSION 3.31)
project(TribitConnect VERSION 0.1.0 LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "Tribit Connect requires Windows.")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")

add_executable(tribit-probe src/probe.cpp)
target_compile_definitions(tribit-probe PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(tribit-probe PRIVATE /W4 /utf-8 /permissive-)
target_link_libraries(tribit-probe PRIVATE Ws2_32 Bthprops)
set_target_properties(tribit-probe PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")

include(CTest)
if(BUILD_TESTING)
  add_executable(tribit-protocol-tests tests/protocol_tests.cpp)
  target_include_directories(tribit-protocol-tests PRIVATE src)
  target_compile_options(tribit-protocol-tests PRIVATE /W4 /utf-8 /permissive-)
  add_test(NAME protocol COMMAND tribit-protocol-tests)
endif()
