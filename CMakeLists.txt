cmake_minimum_required(VERSION 3.31)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/packaging/AppxManifest.xml" TRIBIT_MANIFEST)
string(REGEX MATCH "Version=\"([0-9]+)\\.([0-9]+)\\.([0-9]+)\\.([0-9]+)\"" TRIBIT_VERSION_MATCH "${TRIBIT_MANIFEST}")
if(NOT TRIBIT_VERSION_MATCH)
  message(FATAL_ERROR "The production manifest must contain a four-part version.")
endif()
project(TribitConnect VERSION "${CMAKE_MATCH_1}.${CMAKE_MATCH_2}.${CMAKE_MATCH_3}" LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "Tribit Connect requires Windows.")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")

add_executable(tribit-probe src/probe.cpp)
target_compile_definitions(tribit-probe PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(tribit-probe PRIVATE /W4 /utf-8 /permissive-)
target_link_libraries(tribit-probe PRIVATE Ws2_32 Bthprops)
set_target_properties(tribit-probe PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")

option(TRIBIT_BUILD_APP "Build the companion app and Widgets provider" ON)
if(TRIBIT_BUILD_APP)
  if(NOT EXISTS "${CMAKE_BINARY_DIR}/generated/winrt/Microsoft.Windows.Widgets.Providers.h")
    message(FATAL_ERROR "Run scripts/bootstrap.ps1 to generate the Widgets C++ projections.")
  endif()
  add_executable(TribitConnect WIN32 src/app.cpp packaging/app.manifest packaging/app.rc)
  # The companion window's UI runs in WebView2; the static loader avoids shipping its DLL.
  set(TRIBIT_WEBVIEW2 "${CMAKE_SOURCE_DIR}/.deps/nuget/webview2/build/native")
  if(NOT EXISTS "${TRIBIT_WEBVIEW2}/include/WebView2.h")
    message(FATAL_ERROR "Run scripts/bootstrap.ps1 to fetch the WebView2 SDK.")
  endif()
  target_include_directories(TribitConnect BEFORE PRIVATE "${CMAKE_BINARY_DIR}/generated" "${TRIBIT_WEBVIEW2}/include")
  target_compile_definitions(TribitConnect PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
  target_compile_options(TribitConnect PRIVATE /W4 /utf-8 /permissive- /bigobj)
  target_link_libraries(TribitConnect PRIVATE Ws2_32 Bthprops windowsapp comctl32 dwmapi shell32 advapi32 ole32 uxtheme windowscodecs uuid "${TRIBIT_WEBVIEW2}/x64/WebView2LoaderStatic.lib")
  set_target_properties(TribitConnect PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
  # The one-file installer embeds the signed MSIX, so scripts/package.ps1 builds it after packaging.
  set(TRIBIT_ICON "${CMAKE_SOURCE_DIR}/packaging/app.ico")
  set(TRIBIT_SETUP "${CMAKE_BINARY_DIR}/setup")
  configure_file(packaging/setup.rc.in "${TRIBIT_SETUP}/setup.rc" @ONLY)
  add_executable(TribitConnectSetup EXCLUDE_FROM_ALL WIN32 src/setup.cpp packaging/setup.manifest "${TRIBIT_SETUP}/setup.rc")
  target_include_directories(TribitConnectSetup BEFORE PRIVATE "${CMAKE_BINARY_DIR}/generated")
  target_compile_definitions(TribitConnectSetup PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
  target_compile_options(TribitConnectSetup PRIVATE /W4 /utf-8 /permissive-)
  target_link_libraries(TribitConnectSetup PRIVATE windowsapp comctl32 crypt32 urlmon shell32 ole32)
  set_target_properties(TribitConnectSetup PROPERTIES OUTPUT_NAME "TribitConnect-Setup" RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")

  # Card exporter for the browser mock (scripts/mock.ps1); not part of the package.
  add_executable(widget-mock EXCLUDE_FROM_ALL tools/widget_mock.cpp)
  target_include_directories(widget-mock BEFORE PRIVATE "${CMAKE_BINARY_DIR}/generated" src)
  target_compile_definitions(widget-mock PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
  target_compile_options(widget-mock PRIVATE /W4 /utf-8 /permissive- /bigobj)
  target_link_libraries(widget-mock PRIVATE Ws2_32 Bthprops windowsapp windowscodecs ole32 user32)
  set_target_properties(widget-mock PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
endif()

include(CTest)

  add_executable(tribit-protocol-tests tests/protocol_tests.cpp)
  target_include_directories(tribit-protocol-tests PRIVATE src)
  target_compile_options(tribit-protocol-tests PRIVATE /W4 /utf-8 /permissive-)
  add_test(NAME protocol COMMAND tribit-protocol-tests)
  if(TRIBIT_BUILD_APP)
    add_executable(tribit-core-tests tests/core_tests.cpp)
    target_include_directories(tribit-core-tests BEFORE PRIVATE "${CMAKE_BINARY_DIR}/generated" src)
    target_compile_definitions(tribit-core-tests PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
    target_compile_options(tribit-core-tests PRIVATE /W4 /utf-8 /permissive- /bigobj)
    target_link_libraries(tribit-core-tests PRIVATE Ws2_32 Bthprops windowsapp windowscodecs uuid ole32)
    add_test(NAME core COMMAND tribit-core-tests)
  endif()
endif()
